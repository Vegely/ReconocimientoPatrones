	\begin{frame}{Introducción a la termografía (TIR)}
		% PARTE SUPERIOR: Contexto y Desafíos
		\textbf{Contexto:}
		\begin{itemize}
			\item Visión artificial crítica en seguridad y conducción autónoma.
			\item Imágenes RGB fallan con mala iluminación (niebla, noche).
			\item \textbf{Solución:} Termografía infrarroja (TIR) $\rightarrow$ detecta radiación térmica.
		\end{itemize}
		
		\vspace{0.2cm} % Espacio pequeño entre secciones
		
		\textbf{Desafíos en TIR:}
		\begin{itemize}
			\item Baja resolución (sensores caros), falta de color (monocanal).
			\item Texturas difusas y efecto halo.
		\end{itemize}
		
		\vfill % Empuja el objetivo al fondo
		
		% PARTE INFERIOR: Objetivo (Ancho completo o ajustado)
		\begin{columns}
			\column{0.9\textwidth} % Un poco menos de 1 para que no pegue a los bordes
			\begin{block}{Objetivo}
				Validar la arquitectura \textbf{YOLOv5} sobre el dataset Teledyne FLIR para detección en tiempo real.
			\end{block}
		\end{columns}
	\end{frame}
	
	\begin{frame}{Estado del Arte}
		\begin{columns}
			% Columna izquierda para el texto
			\begin{column}{0.6\textwidth}
				\begin{itemize}
					\item \textbf{Enfoque Clásico:} HOG + SVM. 
					\begin{itemize}
						\item Poca robustez ante variabilidad térmica.

					\end{enumerate}

	\begin{frame}{Ecuaciones del modelo: Convolución y Dimensiones}
		\textbf{Operación de Convolución 2D:}
		\begin{itemize}
			\item Implementada técnicamente como correlación cruzada (sin voltear el kernel):
		\end{itemize}
		
		\begin{equation}
			(W * X)(c_{out}, h, w) = \sum_{l=0}^{C_{in}-1} \sum_{i=0}^{K_H-1} \sum_{j=0}^{K_W-1} W(c_{out}, l, i, j) \cdot X(l, h \cdot S + i \cdot d , w \cdot S + j \cdot d)
		\end{equation}
		
		\vfill
		
		\textbf{Dimensiones del mapa de características de salida:}
		\begin{itemize}
			\item Relación generalizada considerando el factor de dilatación $d$:
		\end{itemize}
		
		\begin{equation}
			Out = \left\lfloor \frac{In + 2P}{S} \right\rfloor + 1
		\end{equation}
	\end{frame}
	
	\begin{frame}{Función de Activación: SiLU}
		\begin{columns}
			\column{0.5\textwidth}
			\textbf{Sigmoid Linear Unit (SiLU):}
			\begin{equation}
				f(z) = z \cdot \sigma(z) = \frac{z}{1+e^{-z}}
			\end{equation}
			\begin{itemize}
				\item Permite flujo de gradientes en valores negativos pequeños.
				\item Suavidad superior a ReLU.
				\item Mejora la convergencia en redes profundas.
			\end{itemize}

	\begin{frame}{Bloque SPPF (Spatial Pyramid Pooling Fast)}
\begin{itemize}
	\item \textbf{Problema:} Capturar contexto multiescala sin el coste computacional de kernels grandes en paralelo.
	\item \textbf{Solución:} Estructura en \textbf{cascada} de Max Pooling ($5\times5$). Tres capas en serie equivalen matemáticamente a un kernel grande.
	\item \textbf{Invarianza:} El Max Pooling extrae la característica dominante, otorgando robustez ante  traslaciones y deformaciones del objeto.
\end{itemize}

	\begin{frame}{Función de Pérdida}
		\textbf{Intuición (YOLOv1):}
		\vspace{-0.1cm}
	\begin{equation*}
	\resizebox{\textwidth}{!}{%
		$ \begin{aligned}
		\mathcal{L} &= \lambda_{\text{coord}} \sum_{i=0}^{S^2} \sum_{j=0}^B \mathds{1}_{i,j}^{\text{obj}} \left[ (x_i - \hat{x}_i)^2 + (y_i - \hat{y}_i)^2 \right] 
		+ \lambda_{\text{coord}} \sum_{i=0}^{S^2} \sum_{j=0}^B \mathds{1}_{i,j}^{\text{obj}} \left[ \left( \sqrt{w_i} - \sqrt{\hat{w}_i} \right)^2 + \left( \sqrt{h_i} - \sqrt{\hat{h}_i} \right)^2 \right] \\
		& + \sum_{i=0}^{S^2} \sum_{j=0}^B \mathds{1}_{i,j}^{\text{obj}} (C_i - \hat{C}_i)^2 
		+ \lambda_{\text{noobj}} \sum_{i=0}^{S^2} \sum_{j=0}^B \mathds{1}_{i,j}^{\text{noobj}} (C_i - \hat{C}_i)^2 
		+ \sum_{i=0}^{S^2} \mathds{1}_i^{\text{obj}} \sum_{c \in \text{classes}}(p_i(c) - \hat{p}_i(c))^2
		\end{aligned} $ 
	} 
	\end{equation*}
		
		\vfill
		
		\textbf{Refinamiento (YOLOv5):}
		% Ecuación en 1 sola línea
		\begin{equation*}
		\begin{aligned}
		\mathcal{L} &= \lambda_{\text{loc}}\,\mathcal{L}_{\text{loc}}
		+
		\lambda_{\text{obj}}\,\mathcal{L}_{\text{obj}}
		+
		\lambda_{\text{cls}}\,\mathcal{L}_{\text{cls}} \\
		&=\lambda_{\text{loc}}(1 - \mathrm{CIoU}) + \lambda_{\text{obj}}\,\mathrm{BCE}_{\text{obj}} + \lambda_{\text{cls}}\,\mathrm{BCE}_{\text{cls}}
		\end{aligned}
		\end{equation*}
	\end{frame}

	% --- Slide 3: Optimización CLR (Warmup + Entrenamiento) ---
	\begin{frame}{Actualización de los pesos}
		\small % Reducimos un poco la fuente global para que quepa todo bien
		La actualización de pesos sigue la regla del \textbf{Descenso del Gradiente Estocástico}, usando \textbf{CLR} como \textit{scheduler} para la tasa de aprendizaje $\varepsilon$:
		\vspace{-0.1cm}
		\begin{equation*}
		\theta_{i+1} = \theta_i - \varepsilon \nabla_{\theta} \mathcal{L}(\theta_i)
		\end{equation*}
		\vspace{-0.3cm}
		
		\begin{columns}[T]
			% --- Columna Izquierda: Fórmulas ---
			\column{0.5\textwidth}
			\textbf{1. Fase de Calentamiento (Warmup):}
			\vspace{-0.2cm}
			\begin{align*}
			\varepsilon &= \varepsilon_0 + \frac{n_i}{n_w} (\varepsilon_1 lr_0 - \varepsilon_0) \\
			\varepsilon_1 &= \left(1 - \frac{ep}{Ep}\right)(1 - lr_f) + lr_f \\
			n_i &= i + ep \cdot n_b \\
			n_w &= \max(\text{\texttt{warmup\_epochs}} \cdot n_b, 100)
			\end{align*}
			
			\vspace{-0.3cm}
			\textbf{2. Fase de Entrenamiento:}
			\vspace{-0.1cm}
			\begin{equation*}
			\varepsilon = \varepsilon_0 \left[\left(1 - \frac{ep}{Ep}\right)(1 - lr_f) + lr_f\right]
			\end{equation*}
			
			% --- Columna Derecha: Intuición e Imagen ---
			\column{0.48\textwidth}
			\vspace{-0.5cm} % Subimos el bloque
			
			\centering
			\vspace{-0.1cm}
			\includegraphics[width=\linewidth, height=3.5cm, keepaspectratio]{clr_vs_fixed.png}
			\captionof{figure}{\scriptsize CLR (rojo) vs Fijo (azul).}
			\vspace{-0.3cm}
			\begin{alertblock}{Intuición}
				\footnotesize
				Usar CLR permite mantener tasas de aprendizaje altas. La convergencia es más rápida y evita mínimos locales.
			\end{alertblock}
		\end{columns}
	\end{frame}

	% --- Slide 4: Dinámica del Optimizador (Momento y L2) ---
	\begin{frame}{Dinámica del Optimizador}
		\begin{columns}[T]
			% --- Columna Izquierda: Todo sobre Momento ---
			\column{0.48\textwidth}
			\begin{block}{Momento de Nesterov}
				\footnotesize
				Iérmino de momento $\mu \cdot v_i$ para evitar mínimos locales y acelerar en zonas planas:
				\vspace{-0.1cm}
				\begin{equation*}
				\begin{cases}
				\theta_{i+1} = \theta_i - \varepsilon v_{i+1} \\
				v_{i+1} = \mu \cdot v_i + \nabla_{\theta}\mathcal{L}(\theta_i - \varepsilon \mu \cdot v_i)
				\end{cases}
				\end{equation*}
				
				\vspace{0.2cm}
				\hrule % Línea separadora sutil
				\vspace{0.2cm}
				
				\textbf{Calendario de Actualización:} \\
				El momento crece linealmente durante el \textit{warmup} hasta fijarse en \texttt{momentum}:
				\vspace{-0.1cm}
				\begin{equation*}
				\mu = \frac{n_i}{n_w}(\text{\texttt{mom}} - \text{\texttt{wup\_mom}}) + \text{\texttt{wup\_mom}}
				\end{equation*}
			\end{block}
			
			% --- Columna Derecha: Todo sobre L2 ---
			\column{0.48\textwidth}
			\begin{alertblock}{Regularización L2 (Weight Decay)}
				\footnotesize
				Penalización cuadrática de los pesos en la pérdida:
				\vspace{-0.1cm}
				\begin{equation*}
				\mathcal{L}_{\text{reg}} = \mathcal{L}(\theta) + \frac{\text{\texttt{wd}}}{2}\sum_i \theta_i^2
				\end{equation*}
				
				\vspace{0.2cm}
				\hrule % Línea separadora sutil
				\vspace{0.2cm}
				
				\textbf{Implementación Eficiente:} \\
				Se integra la penalización en la regla de actualización, evitando la suma de los cuadrados de los pesos (\textit{forward}) y derivar el término del sumatorio (\textit{bacward}):
				\vspace{-0.1cm}
				\begin{equation*}
				\theta_{i+1} = \theta_i (1 - \varepsilon \cdot \text{\texttt{wd}}) - \varepsilon \nabla_{\theta} \mathcal{L}(\theta_i)
				\end{equation*}
			\end{alertblock}
		\end{columns}
	\end{frame}

% --- Diapositiva 1: Métricas Estándar ---
\begin{frame}{Métricas de Desempeño (I): Evaluación Puntual}
	\begin{columns}[t] % [t] alinea las columnas arriba
		
		% Columna Izquierda (Ancho reducido al 48% para dejar margen)
		\column{0.48\textwidth}
		\begin{block}{Precisión ($P$)}
			Fiabilidad de las detecciones positivas.
			\begin{equation*}
			\text{Precisión} = \frac{TP}{TP + FP}
			\end{equation*}
		\end{block}
		
		\vspace{0.3cm} % Espacio vertical para que no se peguen los bloques
		
		\begin{block}{Recall ($R$)}
			Capacidad del modelo para encontrar todos los objetos reales.
			\begin{equation*}
			\text{Recall} = \frac{TP}{TP + FN}
			\end{equation*}
		\end{block}
		
		% Columna Derecha
		\column{0.48\textwidth}
		\begin{block}{F1-Score}
			Media armónica entre Precisión y Recall (balance único).
			\begin{equation*}
			F_1 = 2 \cdot \frac{\text{Precisión} \cdot \text{Recall}}{\text{Precisión} + \text{Recall}}
			\end{equation*}
		\end{block}
		
		\vspace{0.3cm} % Espacio vertical
		
		\begin{block}{IoU Media}
			Calidad media de la localización geométrica de los aciertos.
			\begin{equation*}
			\text{IoU media (TP)} = \frac{1}{TP} \sum \text{IoU}_{\text{TP}}
			\end{equation*}
		\end{block}
	\end{columns}
\end{frame}
% --- Diapositiva 2: Métricas Globales ---
\begin{frame}{Métricas de Desempeño (II): AP y mAP}
	\begin{columns}
		% Columna Izquierda: Definiciones Matemáticas
		\column{0.5\textwidth}
		\begin{block}{Average Precision (AP)}
			Área bajo la curva P-R (aproximación por sumatoria de rectángulos):
			\begin{equation*}
			AP = \sum_k (R_{k+1} - R_k) \cdot P_{k+1}
			\end{equation*}
		\end{block}
		
		\begin{block}{mean Average Precision (mAP)}
			Promedio ponderado por la frecuencia de cada clase ($n_c$) para mitigar el desbalanceo:
			\begin{equation*}
			mAP = \sum_{c=1}^{N} \underbrace{\frac{n_c}{\sum n_k}}_{w_c} \cdot AP_c
			\end{equation*}
		\end{block}
		
		% Columna Derecha: Gráfico Explicativo
		\column{0.5\textwidth}
		\centering
		\begin{tikzpicture}[scale=0.8]

\begin{frame}{Limitaciones y Líneas Futuras}
	\begin{alertblock}{Limitaciones Críticas}
		\begin{itemize}
			\item \textbf{Desbalanceo de Datos:} La detección es casi nula en clases minoritarias (bicicletas, perros) debido a la hegemonía de coches y personas.
			\item \textbf{Ambigüedad Térmica:} Falsos positivos en \textit{Señales} por falta de textura/color.
			\item \textbf{Seguridad:} Se detectaron fallos en peatones vulnerables (niños). No apto aún para autonomía completa sin fusión de sensores.
		\end{itemize}
	\end{alertblock}
	
	\vspace{0.3cm}
	\textbf{Propuestas de Trabajo Futuro:}
	\begin{enumerate}
		\item \textbf{Remuestreo:} Técnicas para equilibrar la distribución de clases.
		\item \textbf{Agrupación Semántica:} Fusión de etiquetas de vehículos grandes.
		\item \textbf{Información Temporal:} Arquitecturas híbridas para evitar pérdidas de detección en fotogramas consecutivos.
	\end{enumerate}
